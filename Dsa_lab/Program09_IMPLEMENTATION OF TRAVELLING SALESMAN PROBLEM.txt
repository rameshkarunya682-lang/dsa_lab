#include <stdio.h>

int cost[10][10], visited[10];
int n, min_cost = 9999;

void tsp(int city, int count, int total_cost)
{
    int i;

    if (count == n)
    {
        if (cost[city][0] != 0)
        {
            total_cost += cost[city][0];

            if (total_cost < min_cost)
                min_cost = total_cost;
        }
        return;
    }

    for (i = 0; i < n; i++)
    {
        if (!visited[i] && cost[city][i] != 0)
        {
            visited[i] = 1;
            tsp(i, count + 1, total_cost + cost[city][i]);
            visited[i] = 0;
        }
    }
}

int main()
{
    int i, j;

    printf("Enter number of cities: ");
    scanf("%d", &n);

    printf("Enter cost matrix:\n");

    for (i = 0; i < n; i++)
    {
        for (j = 0; j < n; j++)
        {
            scanf("%d", &cost[i][j]);
        }
    }

    for (i = 0; i < n; i++)
        visited[i] = 0;

    visited[0] = 1;

    tsp(0, 1, 0);

    printf("Minimum cost: %d", min_cost);

    return 0;
}
